    <script>
        let scene, camera, renderer, sensei, head, students = [];
        let gameActive = false, isWatching = false, score = 0, suspicion = 0;
        let clock = new THREE.Clock();
        let aiTarget = new THREE.Vector3(0, 0, -8);
        let aiState = 'WALK';
        let aiTimer = 0;

        function createDetailedPerson(isSensei = false, clothingColor = 0x2d3436) {
            const group = new THREE.Group();
            const skinMat = new THREE.MeshStandardMaterial({ color: 0xffdbac });
            const clothMat = new THREE.MeshStandardMaterial({ color: clothingColor });
            const torso = new THREE.Group();
            const chest = new THREE.Mesh(new THREE.BoxGeometry(0.55, 0.4, 0.3), clothMat);
            chest.position.y = 1.25;
            const waist = new THREE.Mesh(new THREE.BoxGeometry(0.45, 0.4, 0.25), clothMat);
            waist.position.y = 0.95;
            torso.add(chest, waist);
            group.add(torso);

            if (isSensei) {
                const shirt = new THREE.Mesh(new THREE.BoxGeometry(0.15, 0.3, 0.05), new THREE.MeshStandardMaterial({ color: 0xffffff }));
                shirt.position.set(0, 1.35, 0.15);
                const tie = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.25, 0.02), new THREE.MeshStandardMaterial({ color: 0x991b1b }));
                tie.position.set(0, 1.25, 0.18);
                torso.add(shirt, tie);
            }

            const createArm = (side) => {
                const arm = new THREE.Group();
                const upper = new THREE.Mesh(new THREE.BoxGeometry(0.18, 0.4, 0.18), clothMat);
                upper.position.y = -0.2;
                const lower = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.4, 0.16), clothMat);
                lower.position.y = -0.6;
                arm.add(upper, lower);
                arm.position.set(side * 0.35, 1.45, 0);
                return arm;
            };
            group.armL = createArm(-1);
            group.armR = createArm(1);
            group.add(group.armL, group.armR);

            const createLeg = (side) => {
                const leg = new THREE.Group();
                const thigh = new THREE.Mesh(new THREE.BoxGeometry(0.22, 0.5, 0.22), new THREE.MeshStandardMaterial({ color: 0x1e293b }));
                thigh.position.y = -0.25;
                const shin = new THREE.Mesh(new THREE.BoxGeometry(0.18, 0.5, 0.18), new THREE.MeshStandardMaterial({ color: 0x1e293b }));
                shin.position.y = -0.7;
                leg.add(thigh, shin);
                leg.position.set(side * 0.15, 0.8, 0);
                return leg;
            };
            group.legL = createLeg(-1);
            group.legR = createLeg(1);
            group.add(group.legL, group.legR);

            const headGroup = new THREE.Group();
            const skull = new THREE.Mesh(new THREE.SphereGeometry(0.22, 32, 32), skinMat);
            skull.scale.set(0.9, 1.1, 1);
            headGroup.add(skull);

            if (isSensei) {
                const eyeGeo = new THREE.SphereGeometry(0.025, 8, 8);
                const eyeMat = new THREE.MeshBasicMaterial({ color: 0x000000 });
                const eL = new THREE.Mesh(eyeGeo, eyeMat);
                eL.position.set(-0.08, 0.05, 0.18);
                const eR = new THREE.Mesh(eyeGeo, eyeMat);
                eR.position.set(0.08, 0.05, 0.18);
                headGroup.add(eL, eR);
                group.eyes = [eL, eR];
            } else {
                const hairTypes = [
                    () => {
                        const h = new THREE.Mesh(new THREE.SphereGeometry(0.24, 16, 16, 0, Math.PI * 2, 0, Math.PI / 1.8), new THREE.MeshStandardMaterial({ color: 0x2d3436 }));
                        h.scale.set(1.05, 1.1, 1.1);
                        return h;
                    },
                    () => {
                        const h = new THREE.Mesh(new THREE.SphereGeometry(0.23, 16, 16, 0, Math.PI * 2, 0, Math.PI / 2), new THREE.MeshStandardMaterial({ color: 0x4b3621 }));
                        return h;
                    }
                ];
                headGroup.add(hairTypes[Math.floor(Math.random() * hairTypes.length)]());
            }
            headGroup.position.y = 1.6;
            group.add(headGroup);
            group.head = headGroup;
            return group;
        }

        function init() {
            scene = new THREE.Scene();
            scene.background = new THREE.Color(0xf1f5f9);
            scene.fog = new THREE.Fog(0xf1f5f9, 5, 40);
            camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.set(0, 2.2, 6.5);
            camera.lookAt(0, 1.2, 0);
            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.shadowMap.enabled = true;
            document.body.appendChild(renderer.domElement);
            scene.add(new THREE.AmbientLight(0xffffff, 0.5));
            const light = new THREE.DirectionalLight(0xffffff, 0.8);
            light.position.set(10, 20, 10);
            light.castShadow = true;
            scene.add(light);

            const floor = new THREE.Mesh(new THREE.PlaneGeometry(100, 100), new THREE.MeshStandardMaterial({ color: 0xe2e8f0 }));
            floor.rotation.x = -Math.PI / 2;
            floor.receiveShadow = true;
            scene.add(floor);

            for (let x = -8; x <= 8; x += 4) {
                for (let z = -12; z <= 4; z += 4) {
                    const desk = new THREE.Group();
                    const top = new THREE.Mesh(new THREE.BoxGeometry(2.5, 0.1, 1.5), new THREE.MeshStandardMaterial({ color: 0xffffff }));
                    top.position.y = 0.9;
                    top.castShadow = true;
                    top.receiveShadow = true;
                    desk.add(top);
                    desk.position.set(x, 0, z);
                    scene.add(desk);
                    if (x !== 0 || z !== 4) {
                        const s = createDetailedPerson(false, Math.random() * 0xffffff);
                        s.position.set(x, 0, z + 1);
                        s.rotation.y = Math.PI;
                        s.head.rotation.x = 0.6;
                        scene.add(s);
                        students.push(s);
                    }
                }
            }

            sensei = createDetailedPerson(true, 0x1e293b);
            sensei.position.set(0, 0, -10);
            scene.add(sensei);

            const startAction = (e) => {
                if (!gameActive) return;
                isWatching = true;
                document.getElementById('screen-study').classList.add('hidden');
                document.getElementById('screen-youtube').classList.remove('hidden');
                document.getElementById('youtube-device').classList.add('active-view');
            };
            const stopAction = (e) => {
                isWatching = false;
                document.getElementById('screen-study').classList.remove('hidden');
                document.getElementById('screen-youtube').classList.add('hidden');
                document.getElementById('youtube-device').classList.remove('active-view');
            };
            window.addEventListener('mousedown', startAction);
            window.addEventListener('mouseup', stopAction);
            window.addEventListener('touchstart', startAction);
            window.addEventListener('touchend', stopAction);
            animate();
        }

        function updateAI() {
            const dt = clock.getDelta();
            const time = Date.now() * 0.005;
            aiTimer -= 1;
            if (aiState === 'WALK') {
                let dist = sensei.position.distanceTo(aiTarget);
                if (dist > 0.2) {
                    let dir = aiTarget.clone().sub(sensei.position).normalize();
                    sensei.position.add(dir.multiplyScalar(0.04));
                    sensei.rotation.y = THREE.MathUtils.lerp(sensei.rotation.y, Math.atan2(dir.x, dir.z), 0.1);
                    sensei.legL.rotation.x = Math.sin(time) * 0.45;
                    sensei.legR.rotation.x = Math.cos(time) * 0.45;
                    sensei.armL.rotation.x = Math.cos(time) * 0.3;
                    sensei.armR.rotation.x = Math.sin(time) * 0.3;
                } else {
                    aiState = 'INSPECT';
                    aiTimer = 80 + Math.random() * 150;
                }
            } else if (aiState === 'INSPECT') {
                sensei.legL.rotation.x = 0;
                sensei.legR.rotation.x = 0;
                sensei.armL.rotation.x = 0.1;
                sensei.armR.rotation.x = 0.1;
                sensei.head.rotation.y = Math.sin(time * 0.4) * 1.3;
                if (aiTimer <= 0) {
                    aiState = 'WALK';
                    aiTarget.set((Math.random() - 0.5) * 14, 0, (Math.random() - 0.5) * 20 - 4);
                }
            }

            const playerPos = new THREE.Vector3(0, 0, 5);
            const dist = sensei.position.distanceTo(playerPos);
            const forward = new THREE.Vector3(0, 0, 1).applyQuaternion(sensei.head.getWorldQuaternion(new THREE.Quaternion()));
            const toPlayer = playerPos.clone().sub(sensei.position).normalize();
            const dot = forward.dot(toPlayer);

            if (isWatching) {
                score += 2;
                document.getElementById('score-text').innerText = score.toString().padStart(4, '0');
                document.getElementById('yt-progress').style.width = (score / 10 % 100) + "%";
                if (dot > 0.45 && dist < 11) {
                    suspicion += 2.2;
                    document.getElementById('warning-vignette').style.opacity = "1";
                    sensei.eyes.forEach(e => e.material.color.set(0xff0000));
                } else {
                    suspicion = Math.max(0, suspicion - 0.2);
                    document.getElementById('warning-vignette').style.opacity = "0";
                    sensei.eyes.forEach(e => e.material.color.set(0x000000));
                }
            } else {
                suspicion = Math.max(0, suspicion - 1.2);
                document.getElementById('warning-vignette').style.opacity = "0";
                sensei.eyes.forEach(e => e.material.color.set(0x000000));
            }
            document.getElementById('suspicion-bar').style.width = suspicion + "%";
            if (suspicion >= 100) gameOver();
        }

        function animate() {
            requestAnimationFrame(animate);
            if (gameActive) updateAI();
            renderer.render(scene, camera);
        }

        function gameOver() {
            gameActive = false;
            document.getElementById('start-screen').style.display = 'flex';
            document.getElementById('start-screen').querySelector('h1').innerText = "DISMISSED";
            document.getElementById('start-screen').querySelector('p').innerHTML = `<span class='text-red-500 font-black text-2xl'>Score: ${score}</span><br><br>大網先生「...今の動画、面白かったか？」`;
            document.getElementById('start-btn').innerText = "反省して再入室";
            isWatching = false;
        }

        document.getElementById('start-btn').onclick = () => {
            score = 0;
            suspicion = 0;
            gameActive = true;
            document.getElementById('start-screen').style.display = 'none';
        };

        window.onresize = () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        };
        window.onload = init;
    </script>
